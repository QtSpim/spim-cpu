cmake_minimum_required(VERSION 3.10.0)
project(spim-cpu VERSION 9.1.24 LANGUAGES CXX)

# Library declaration
##############################################################################
add_library(spim-cpu)

# Flex and Bison targets
##############################################################################
# Note:
#   - COMPILE_FLAGS in flex_target() and bison_target() is deprecated in
#     CMake 4.0 and later in favour of OPTIONS;
#     however, OPTIONS is not available in CMake versions below 4.0. These
#     should be updated to use OPTIONS once CMake 4.0 is widely available.
#
#   - The header generated by DEFINES_FILE is placed in the same directory as
#     the generated parser source file because the generated source has a
#     #include directive that expects the header to be in the same directory.
#     We manually copy the generated header to the include directory so that
#     it can be included as <spim-cpu/parser-yacc.h> by other source files and
#     by users of the library.
add_custom_target(create_flex_bison_output_dirs
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/include/spim-cpu
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/src
)
add_dependencies(spim-cpu create_flex_bison_output_dirs)

find_package(FLEX REQUIRED)
flex_target(scanner
  src/scanner.l                                               # Input file
  ${CMAKE_CURRENT_BINARY_DIR}/src/scanner.yy.cpp              # Output file
  COMPILE_FLAGS "-I -8"
)

find_package(BISON REQUIRED)
bison_target(parser
  src/parser.y                                                # Input file
  ${CMAKE_CURRENT_BINARY_DIR}/src/parser-yacc.cpp             # Output file
  DEFINES_FILE ${CMAKE_CURRENT_BINARY_DIR}/src/parser-yacc.h  # Header file for token definitions
  COMPILE_FLAGS "-p yy"
)

add_flex_bison_dependency(scanner parser)

add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/include/spim-cpu/parser-yacc.h
  COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_BINARY_DIR}/src/parser-yacc.h ${CMAKE_CURRENT_BINARY_DIR}/include/spim-cpu/parser-yacc.h
  DEPENDS ${BISON_parser_OUTPUTS} create_flex_bison_output_dirs
)

# Sources and compiler flags
##############################################################################
target_sources(spim-cpu
  # Implementation files and private headers (not installed)
  PRIVATE
    src/data.cpp src/display-utils.cpp src/inst.cpp
    src/mem.cpp src/run.cpp src/spim-utils.cpp src/string-stream.cpp
    src/sym-tbl.cpp src/syscall.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/src/scanner.yy.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/src/parser-yacc.cpp    

  PRIVATE
    FILE_SET private_headers
    TYPE HEADERS
    BASE_DIRS ${CMAKE_CURRENT_BINARY_DIR}/src
    FILES
      ${CMAKE_CURRENT_BINARY_DIR}/src/parser-yacc.h
    BASE_DIRS src
    FILES
      src/op.h  # While this file is available to .cpp files in src
                # even without being listed here, it is not available
                # to scanner.l since Flex runs in the binary directory
                # (CMAKE_CURRENT_BINARY_DIR).

  # Headers available to both implementation and users of the library
  PUBLIC
    FILE_SET HEADERS
    BASE_DIRS include # So that headers are included as <spim-cpu/header.h>
    FILES
      include/spim-cpu/data.h include/spim-cpu/inst.h
      include/spim-cpu/mem.h
      include/spim-cpu/parser.h include/spim-cpu/reg.h
      include/spim-cpu/run.h include/spim-cpu/scanner.h
      include/spim-cpu/spim-syscall.h include/spim-cpu/spim-utils.h
      include/spim-cpu/spim.h include/spim-cpu/string-stream.h
      include/spim-cpu/sym-tbl.h include/spim-cpu/syscall.h
      include/spim-cpu/version.h
    BASE_DIRS ${CMAKE_CURRENT_BINARY_DIR}/include
    FILES
      ${CMAKE_CURRENT_BINARY_DIR}/include/spim-cpu/parser-yacc.h
)

target_compile_definitions(spim-cpu
  PUBLIC
    TEXT_SIZE=65536
    DATA_SIZE=131072
    K_TEXT_SIZE=65536
)

target_compile_options(spim-cpu
  PUBLIC
    -Wall -pedantic -Wextra -Wunused -Wno-write-strings
)
